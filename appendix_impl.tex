\section{Windowing and Monitoring}
\begin{itemize}
    \item 	extbf{Reference Window ($W_{ref}$):} 500 samples, maintained via Reservoir Sampling \citep{Vitter1985}.
    \item 	extbf{Current Window ($W_{cur}$):} 200 samples, sliding window.
    \item 	extbf{Monitoring Frequency:} Every 50 samples.
\end{itemize}

\section{Uncertainty and Statistical Testing}
\begin{itemize}
    \item 	extbf{Prediction Uncertainty Index (PUI):} Entropy-based warning.
    \item 	extbf{Adaptive Sequential Permutation Test (ASPT):} Max 50 perms, $\alpha=0.01$.
\end{itemize}

\section{Drift Severity Index (DSI) Tiers}
\begin{itemize}
    \item 	extbf{Critical (DSI $> 0.6$):} concentrated univariate.
    \item 	extbf{Moderate (DSI $0.3$--$0.6$):} subset drift.
    \item 	extbf{Low (DSI $< 0.3$):} diffuse shift.
\end{itemize}
